# --- 2. Quiet scope. Since the 2026-09-24 boot, the managed dolt watchdog and server run inside the
# supervisor unit's cgroup (the watchdog is parented by the user manager, the server by the watchdog).
# Sequence 13 required the supervisor to be alone there. Exactly these two members are admitted,
# identified by exact argv and executable, and nothing else.
def s14_proc(pid):
    argv=Path('/proc',str(pid),'cmdline').read_bytes().split(b'\0')[:-1]
    exe=os.readlink('/proc/'+str(pid)+'/exe')
    return [a.decode() for a in argv],exe

def s14_quiet_scope(host):
    o,s=c.o,c.s
    pid=host['host']['pid']
    first={n:s.members(n) for n in (o.SERVICE,o.RECONCILER)}
    table=s.process_table()
    o.require(first[o.RECONCILER]==set(),'reconciler scope residue')
    others=sorted(first[o.SERVICE]-{pid})
    o.require(pid in first[o.SERVICE] and len(others)==2,'supervisor scope membership')
    ident={p:s14_proc(p) for p in others}
    watchdog=[p for p,(argv,_) in ident.items() if argv[:2]==[S14_GC_BIN,'__gc-managed-dolt-scope-watchdog']]
    server=[p for p,(argv,_) in ident.items() if argv[:2]==['dolt','sql-server']]
    o.require(len(watchdog)==1 and len(server)==1,'dolt scope members')
    w,sv=watchdog[0],server[0]
    o.require(ident[w]==([S14_GC_BIN,'__gc-managed-dolt-scope-watchdog',S14_DOLT_CFG,S14_DOLT_LOG,str(S14_CITY)],S14_GC_BIN),
              'dolt watchdog identity')
    o.require(ident[sv]==(['dolt','sql-server','--config',S14_DOLT_CFG],S14_DOLT_BIN),'dolt server identity')
    o.require(table[sv][0]==w,'dolt server parent')
    o.require(s.descendants(table,{pid})=={pid},'supervisor descendant residue')
    o.require(s.descendants(table,{w})=={w,sv},'dolt descendant residue')
    second=s.process_table()
    o.require(all(table.get(p)==second.get(p) for p in (pid,w,sv)) and first=={n:s.members(n) for n in first},

# --- 3. Cache admission. After the new Core starts, exactly one new synthetic directory (S14_NEW_KEY)
# must appear, and S14_OPTIONAL_KEY may appear. Their full contents must equal the S1b expectation,
# with the marker derived per key. Every pre-existing entry stays exact, atime included. Only the
# cache root entry's mtime, ctime and nlink may change.
def s14_expected_files():
    data=Path(S14_EXPECTATION).read_bytes()
    c.o.require(s14_sha(data)==S14_EXPECTATION_SHA,'cache expectation drift')
    return {f['Path']:f for f in json.loads(data)['new']['materialized_files']}

def s14_marker(commit):
    return ('schema = 1\nrepository = "https://github.com/gastownhall/gascity.git"\n'
            'commit = "'+commit+'"\ncontent_hash = "'+S14_NEW_HASH+'"\n').encode()

def s14_check_new_dir(key):
    expected=s14_expected_files()
    root=S14_CACHE/key
    seen=set()
    for dirpath,dirnames,filenames in os.walk(root):
        rel_dir=os.path.relpath(dirpath,root)
        for name in list(dirnames)+filenames:
            path=Path(dirpath,name);rel=os.path.normpath(os.path.join(rel_dir,name))
            st=os.lstat(path);seen.add(rel)
            want=expected.get(rel);c.o.require(want is not None,'unexpected cache entry '+key+'/'+rel)
            mode=int(want['Mode'],8)
            if _stat.S_ISDIR(st.st_mode):
                c.o.require(mode&(1<<31) and _stat.S_IMODE(st.st_mode)==mode&0o777,'cache dir mode '+rel)
                continue
            c.o.require(_stat.S_ISREG(st.st_mode) and _stat.S_IMODE(st.st_mode)==mode&0o777,'cache file mode '+rel)
            fd=os.open(path,os.O_RDONLY|os.O_NOATIME|os.O_NOFOLLOW|os.O_CLOEXEC)
            try:
                data=b''
                while True:
                    part=os.read(fd,1048576)
                    if not part:break
                    data+=part
            finally:os.close(fd)
            if rel=='.gc-bundled-pack-cache.toml':
                c.o.require(data==s14_marker(S14_COMMIT_BY_KEY[key]),'cache marker '+key)
            else:
                c.o.require(len(data)==want['Size'] and s14_sha(data)==want['SHA256'],'cache content '+key+'/'+rel)
    c.o.require(seen=={p for p in expected if p!='.'},'cache entry set '+key)

def s14_admit_cache(before,after):
    b,a=before['inventory'],after['inventory']
    c.o.require(set(b)<=set(a),'cache entry removed')
    for key,meta in b.items():
        if key=='.':
            strip=lambda m:{k:v for k,v in m.items() if k not in ('mtime_ns','ctime_ns','nlink','atime_ns')}
            c.o.require(strip(a[key])==strip(meta),'cache root metadata')
        else:
            c.o.require(a[key]==meta,'pre-existing cache entry changed: '+key)
    added=set(a)-set(b)
    tops={k.split('/',1)[0] for k in added}
    c.o.require(S14_NEW_KEY in tops and tops<={S14_NEW_KEY,S14_OPTIONAL_KEY},'unexpected cache additions')
    for key in sorted(tops):
        s14_check_new_dir(key)
    return sorted(tops)

# --- 4. Capture. This is the base capture() with the stale r5 manifest reads removed (the
# preimages come from block 1) and with a candidate cache admitted through block 3 instead of
# required equal.
def s14_capture(candidate):
    o,s=c.o,c.s
    expected=c.expected_predecessor()
    o.GC_SHA=c.NEW if candidate else c.OLD
    host=o.host_observation()
    if not candidate:
        s.broker_precondition(host)
    else:
        service=host['broker']
        o.require(service['ActiveState']=='active' and service['SubState']=='running'
                  and service['NRestarts']=='0' and service['Result']=='success'
                  and service['UnitFileState']=='disabled' and int(service['MainPID'])>1,'broker post-activation state')
        o.require(host['broker_socket']['ActiveState']=='active','broker socket inactive')
    s.preimages(None)
    links=expected['links']
    for path,target in links.items():
        o.require(os.path.islink(path) and os.readlink(path)==target,'link changed')
    pins={}
    for path,wanted in expected['pins'].items():
        if candidate and path==o.GC:
            wanted=dict(wanted,sha256=c.NEW,size=c.NEW_SIZE)
        actual=s.pin(path,links)
        o.require(actual==wanted,'infrastructure pin changed: '+path)
        pins[path]=actual
    suspension=s.pin(o.CITY/'.gc/runtime/suspension-state.json',links)
    o.require(suspension==expected['suspension'],'suspension changed')
    trees,cache,admitted={},None,None
    for path,wanted in expected['trees'].items():
        is_cache=path==str(S14_CACHE)
        value=o.tree_snapshot(path,cache=is_cache)
        summary={k:value[k] for k in ('sha256','entries','file_bytes')}
        if is_cache:
            cache=value
            if candidate:
                admitted=s14_admit_cache(expected['cache'],value)
            else:
                o.require(summary==wanted,'tree changed: '+path)
        else:
            o.require(summary==wanted,'tree changed: '+path)
        trees[path]=summary
    if not candidate:
        o.require(cache==expected['cache'],'cache inventory changed')
    protected={}
    for path,wanted in expected['protected'].items():
        value=o.tree_snapshot(path,protected=True)
        o.require(value==wanted,'protected inventory changed: '+path)
        protected[path]=value
    scope=s.quiet_scope(host)
    s.preimages(None)
    o.require(o.host_observation()==host,'host changed during observation')
    observed=dict(host=host,pins=pins,suspension=suspension,trees=trees,cache=cache,
                  protected=protected,links=links,scope=scope)
    if not candidate:
        o.require(observed==expected,'predecessor closure changed')
    else:
        observed['admitted_cache_keys']=admitted
    return observed
c.capture=s14_capture

# --- 5. Transition. This is sequence 13's rule (fresh Core epoch, the gc pin at NEW) plus fresh dolt
# members with the same identities, the admitted cache, and its tree summary.
def validate_successor_transition(before,after):
    adjusted=copy.deepcopy(before)
    prior,current=before['host'],after['host']
    for key in ('pid','start','listener'):
        c.o.require(current['host'][key]!=prior['host'][key],'Core host epoch not fresh '+key)
        adjusted['host']['host'][key]=current['host'][key]
    core=current['core'];old=prior['core']
    c.o.require(int(core['MainPID'])>1 and core['MainPID']!=old['MainPID'],'Core PID not fresh')
    c.o.require(int(core['ExecMainStartTimestampMonotonic'])>int(old['ExecMainStartTimestampMonotonic']),'Core start not fresh')
    for key in ('MainPID','ExecMainStartTimestampMonotonic'):
        adjusted['host']['core'][key]=core[key]
    adjusted['pins'][c.o.GC]=dict(adjusted['pins'][c.o.GC],sha256=c.NEW,size=c.NEW_SIZE)
    new_scope=after['scope']
    for role in ('watchdog','server'):
        c.o.require(new_scope['dolt_members'][role]!=before['scope']['dolt_members'][role],'dolt member not fresh '+role)
    adjusted['scope']=dict(adjusted['scope'],core_members=new_scope['core_members'],dolt_members=new_scope['dolt_members'])
    cache_path=str(S14_CACHE)
    adjusted['cache']=after['cache'];adjusted['trees'][cache_path]=after['trees'][cache_path]
    adjusted['admitted_cache_keys']=after.get('admitted_cache_keys')
    c.o.require(adjusted==after,'unexpected full-closure transition')

def s14_accept():
    c.o.require(sys.flags.isolated and sys.flags.dont_write_bytecode,'use python3 -I -B')
    S14_ACCEPT_ROOT.mkdir(mode=0o700)
    def rec(name,value):
        with (S14_ACCEPT_ROOT/name).open('x') as stream:
            json.dump(value,stream,indent=1,sort_keys=True)
        return s14_sha((S14_ACCEPT_ROOT/name).read_bytes())
    raw=S14_R7.read_bytes();c.o.require(s14_sha(raw)==S14_R7_SHA,'R7 template drift')
    template=json.loads(raw)['closure']
    rec('start.json',dict(executor_sha256=s14_sha(Path(__file__).read_bytes()),r7_sha256=S14_R7_SHA))
    s14_city_rules(False)
    first=s14_observe_closure(template)
    first_sha=rec('first.json',dict(closure=first,monotonic_ns=time.monotonic_ns()))
    time.sleep(5)
    second=s14_observe_closure(template)
    second_sha=rec('second.json',dict(closure=second,monotonic_ns=time.monotonic_ns()))
    c.o.require(second==first,'predecessor not stable across 5 s')
    s14_city_rules(False)
    changed={k:sorted(p for p in set(template[k])|set(second[k]) if template[k].get(p)!=second[k].get(p))
             for k in ('links','pins','trees','protected')}
    rec('delta-vs-r7.json',dict(changed=changed,suspension_changed=template['suspension']!=second['suspension'],
                               host_keys=sorted(k for k in second['host'] if second['host'][k]!=template['host'].get(k))))
    rec('pass.json',dict(ok=True,first_sha256=first_sha,second_sha256=second_sha,review_required=True))
    print(json.dumps(dict(ok=True,second_sha256=second_sha,
                          changed={k:len(v) for k,v in changed.items()}),sort_keys=True))

if len(sys.argv)==2 and sys.argv[1]=='accept' and __name__=='__main__':
