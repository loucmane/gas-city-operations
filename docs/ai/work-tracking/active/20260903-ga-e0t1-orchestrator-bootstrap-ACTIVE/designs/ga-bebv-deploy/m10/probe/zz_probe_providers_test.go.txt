package config

import (
	"encoding/json"
	"os"
	"sort"
	"strings"
	"testing"

	"github.com/gastownhall/gascity/internal/fsys"
	"github.com/gastownhall/gascity/internal/pathutil"
)

// Throwaway ga-bebv S3 design probe; not committed. Reads CITY_TOML, resolves every agent and every named
// provider, and writes a JSON report to PROBE_OUT.
func TestZZProbeResolvedProviders(t *testing.T) {
	path := os.Getenv("CITY_TOML")
	out := os.Getenv("PROBE_OUT")
	if path == "" || out == "" {
		t.Skip("no CITY_TOML")
	}
	t.Setenv("GC_HOME", os.Getenv("PROBE_GC_HOME"))
	cfg, _, err := LoadWithIncludes(fsys.OSFS{}, path)
	if err != nil {
		t.Fatal(err)
	}
	if replacement := os.Getenv("PROBE_PROVIDERS_TOML"); replacement != "" {
		data, err := os.ReadFile(replacement)
		if err != nil {
			t.Fatal(err)
		}
		parsed, err := Parse(data)
		if err != nil {
			t.Fatal(err)
		}
		for name, spec := range parsed.Providers {
			cfg.Providers[name] = spec
		}
	}
	look := func(name string) (string, error) { return "/usr/bin/" + name, nil }
	type choice struct {
		Value string   `json:"value"`
		Flags []string `json:"flags"`
	}
	type view struct {
		Provider string              `json:"provider"`
		Defaults map[string]string   `json:"defaults"`
		Schema   map[string][]choice `json:"schema"`
		Command  string              `json:"command"`
		Error    string              `json:"error,omitempty"`
		Unsafe   []string            `json:"unsafe_choices,omitempty"`
		Roots    []string            `json:"work_dir_roots,omitempty"`
		WorkDir  string              `json:"work_dir,omitempty"`
		Accepted int                 `json:"guard_accepted,omitempty"`
	}
	render := func(a *Agent) view {
		rp, err := ResolveProvider(a, &cfg.Workspace, cfg.Providers, look)
		if err != nil {
			return view{Error: err.Error()}
		}
		v := view{Provider: rp.Name, Defaults: rp.EffectiveDefaults, Schema: map[string][]choice{}}
		for _, opt := range rp.OptionsSchema {
			for _, c := range opt.Choices {
				v.Schema[opt.Key] = append(v.Schema[opt.Key], choice{c.Value, c.FlagArgs})
				joined := strings.Join(c.FlagArgs, " ")
				if strings.Contains(joined, "dangerously") || strings.Contains(joined, "--yolo") || strings.Contains(joined, "danger-full-access") {
					v.Unsafe = append(v.Unsafe, opt.Key+"="+c.Value)
				}
			}
		}
		if cmd, err := BuildProviderLaunchCommand("", rp, nil, ""); err == nil {
			v.Command = cmd.Command
		} else {
			v.Error = "launch: " + err.Error()
		}
		return v
	}
	report := map[string]view{}
	for i := range cfg.Agents {
		v := render(&cfg.Agents[i])
		v.Roots = cfg.Agents[i].WorkDirRoots
		v.WorkDir = cfg.Agents[i].WorkDir
		for _, root := range v.Roots {
			entries, _ := os.ReadDir(root)
			for _, e := range entries {
				if !e.IsDir() {
					continue
				}
				candidate := root + "/" + e.Name()
				if _, ok := pathutil.GuardLaunchWorkDir(candidate, v.Roots, ""); !ok {
					v.Unsafe = append(v.Unsafe, "guard-refused:"+candidate)
				} else {
					v.Accepted++
				}
			}
		}
		report["agent:"+cfg.Agents[i].QualifiedName()] = v
	}
	names := make([]string, 0, len(cfg.Providers))
	for name := range cfg.Providers {
		names = append(names, name)
	}
	sort.Strings(names)
	for _, name := range names {
		report["provider:"+name] = render(&Agent{Name: "probe", Provider: name})
	}
	data, _ := json.MarshalIndent(report, "", " ")
	if err := os.WriteFile(out, data, 0o600); err != nil {
		t.Fatal(err)
	}
}
